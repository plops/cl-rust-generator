mit diesem genre (aber instrumental):

Drum and Bass at 174 BPM with rolling jump-up momentum, dark jungle reese architecture, distorted 808 sub pressure, crisp fractured breaks, clipped rave-siren transitions, half-time reflective bridge, tightened studio mix and maximum final-tearout drop

To build a standalone Drum and Bass engine in Rust that talks directly to ALSA on Linux, you will use [FunDSP](https://github.com/SamiPerttu/fundsp) to code the synthesis audio graph and [CPAL (Cross-Platform Audio Library)](https://docs.rs/cpal) to handle the native ALSA hardware stream. [1, 2] 
## Project Setup
First, create a new Rust binary project and install the dependencies. In your Linux terminal, run:

cargo new dark_dnb_synth
cd dark_dnb_synth
cargo add fundsp cpal

Ensure your system has the ALSA development headers installed. On Debian/Ubuntu-based systems, run sudo apt install libasound2-dev.
## The Rust Implementation
Replace the contents of src/main.rs with the following code. This script builds a 174 BPM master clock, a detuned, heavily saturated Reese Bass using fundsp math operators, a synchronized Drum Step-Sequencer mapping kick/snare triggers, and pipes the output into your native ALSA system device. [1, 2] 

use cpal::traits::{DeviceTrait, HostTrait, StreamTrait};use fundsp::hacker::*;use std::sync::Arc;use std::thread;use std::time::Duration;
fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 1. Initialise the ALSA host via CPAL
    let host = cpal::default_host();
    
    // Explicitly target the default ALSA output hardware
    let device = host
        .default_output_device()
        .expect("Failed to find default ALSA output device");
    
    let config = device.default_output_config()?;
    let sample_rate = config.sample_rate().0 as f64;

    println!("Playing Dark DnB on ALSA device: {}", device.name()?);
    println!("Tempo: 174 BPM | Sample Rate: {} Hz", sample_rate);

    // 2. Define the Dark DnB Audio Graph using FunDSP
    // Base 174 BPM Clock setup (174 BPM = 2.9 Hz quarter notes, 11.6 Hz for 16th notes)
    let dnb_tempo = 174.0;
    let step_hz = (dnb_tempo / 60.0) * 4.0; // 16th note triggers

    // --- BASS LAYER: Detuned Reese Bass + Tanh Distortion ---
    // Two detuned sawtooth waves fighting in the sub-lows
    let reese_oscillators = lfo(move |_t| 50.0) >> saw() 
                          & lfo(move |_t| 50.8) >> saw();
    
    // Low pass filter + raw heavy saturation for the gritty UK sound system edge
    let reese_bass = reese_oscillators 
        >> lowpass_hz(280.0, 1.0) 
        >> shape(Shape::Tanh(2.5)) * 0.4;

    // --- DRUM LAYER: Algorithmic Synthesized Step-Sequencer ---
    // A 16-step rhythmic pattern loop for Kicks (1) and Snares (2)
    // 1000 0010 0010 0100 (Classic driving DnB step variations)
    let drum_pattern = [1, 0, 0, 0,  0, 0, 2, 0,  0, 0, 2, 0,  0, 1, 0, 0];

    let drum_sequencer = lfo(move |t| {
        let step = (t * step_hz).floor() as usize % 16;
        let trigger_type = drum_pattern[step];
        
        // Simple decay envelope trigger simulation based on clock phase
        let phase = (t * step_hz) % 1.0;
        let env = (-8.0 * phase).exp(); // Fast exponential decay

        if trigger_type == 1 {
            // Kick Drum sound: Pitch-sweeping low sine wave
            let pitch_sweep = 150.0 * env + 45.0;
            (f32::sin((t * pitch_sweep * 2.0 * std::f64::consts::PI) as f32) * env as f32 * 0.7) as f64
        } else if trigger_type == 2 {
            // Snare Drum sound: White noise filtered around mid frequencies
            let noise_sample = (rand::random::<f32>() - 0.5) * 2.0;
            (noise_sample * env as f32 * 0.35) as f64
        } else {
            0.0
        }
    });

    // --- COMBINE AND MASTER: Mix and output to Stereo ---
    let master_mix = (reese_bass + drum_sequencer) >> split::<U2>();

    // 3. Instantiate and run the real-time DSP backend
    let mut audio_node = Block::new(Box::new(master_mix));
    audio_node.set_sample_rate(sample_rate);

    // Build the thread-safe processing loop
    let audio_node = Arc::new(std::sync::Mutex::new(audio_node));
    let stream_node = Arc::clone(&audio_node);

    let stream = device.build_output_stream(
        &config.into(),
        move |data: &mut [f32], _: &cpal::OutputCallbackInfo| {
            let mut node = stream_node.lock().unwrap();
            for frame in data.chunks_mut(2) {
                let mut left = 0.0;
                let mut right = 0.0;
                node.process(1, &[], &mut [left, right]);
                
                if frame.len() == 2 {
                    frame[0] = left as f32;
                    frame[1] = right as f32;
                }
            }
        },

        |err| eprintln!("ALSA Stream Error: {}", err),
        None
    )?;

    // Start playing audio
    stream.play()?;

    // Keep execution alive on the main thread
    println!("Press Ctrl+C to stop playback...");
    loop {
        thread::sleep(Duration::from_millis(100));
    }
}

lange quelltext-dateien sind zu vermeiden. sobald eine datei mehr als eine klar abgrenzbare
zustaendigkeit hat oder ca. 300 zeilen ueberschreitet, ist sie in semantisch getrennte einzelne
dateien zu unterteilen. jede neue datei bekommt am anfang eine zweistellige nummer plus einen
namen der widerspiegelt was darinsteht, z.b. `01_clock.rs`, `02_usb_cdc.rs`, `03_router.rs`
in `firmware/src/` bzw. `01_types.rs`, `02_frame.rs` in `common/src/`. regeln: aufsteigende
nummern folgen der initialisierungs-/datenfluss-reihenfolge (clock vor peripherie, transport
vor protokoll, protokoll vor modus), zusammengehoeriges bleibt in einer datei (kein
auseinanderreissen von zusammengehoerigen typen + ihren tests), `mod.rs`/`lib.rs`/`main.rs`
enthalten nur modul-deklarationen und die oberste verdrahtung (init, spawn, join), kein
verhalten. aufteilung immer ohne verhaltens-aenderung durchfuehren (vorher/nachher: alle tests
plus smoke gruens). bestehende dateien (`main.rs`, `text.rs`, `frame.rs`, `router.rs`) bei
bedarf nach diesen regeln umbauen, sobald du sie anfasst.
